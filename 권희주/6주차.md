## 📒 개요
- 한빛 마켓의 패션 럭키백 상황을 예시로 들어 패션 MNIST 데이터를 분류하며 딥러닝에 대해 공부했다.
- 인공 신경망의 개념과 구조를 배우고, 층을 여러 개 쌓은 심층 신경망을 만들어 보며, 이를 케라스(Keras)의 어떤 함수로 구현하고 훈련 과정을 어떻게 조절할 수 있는지 학습했다.

## 👀 새롭게 배운 점
- 인공 신경망은 입력층, 출력층으로 이루어지며, 케라스의 Dense 클래스로 층을 만들고 Sequential 클래스로 모델을 만든 뒤 compile()로 손실 함수를 지정하고 fit()으로 훈련할 수 있다는 것을 알게 되었다. 또한, 다중 분류에서 타깃이 정수일 때는 sparse_categorical_crossentropy 손실 함수를 사용한다는 것을 알게 되었다.
- 입력층과 출력층 사이에 은닉층을 추가하면 심층 신경망이 되며, 은닉층에는 sigmoid나 relu 같은 활성화 함수를 적용해야 한다는 것을 알게 되었다. 더불어, Flatten 층으로 2차원 이미지를 1차원으로 펼칠 수 있고, optimizer 매개변수로 adam 같은 옵티마이저를 지정해 학습 방식을 바꿀 수 있다는 것을 알게 되었다.
- fit()이 반환하는 history로 손실 곡선을 그려 과대적합을 확인할 수 있으며, 드롭아웃으로 일부 뉴런을 꺼서 과대적합을 줄이고, ModelCheckpoint와 EarlyStopping 콜백으로 가장 좋은 모델을 저장하고 조기 종료할 수 있다는 것을 알게 되었다.

## 💡 느낀 점 (더 알아보고 싶은 내용, 어려웠던 부분 등)
- 4장에서 배운 로지스틱 회귀, 소프트맥스, 확률적 경사 하강법, 조기 종료가 신경망에서도 그대로 쓰여서 앞에서 배운 내용이 연결되는 느낌이 들어 이해에 도움이 됐다. 하지만, 은닉층의 뉴런 개수나 드롭아웃 비율, 옵티마이저 종류처럼 직접 정해야 하는 값이 많아서 어떤 기준으로 정해야 하는지는 아직 잘 모르겠고 더 알아보고 싶다.

```python
"""# 인공 신경망"""
"""## 패션 MNIST"""

from tensorflow import keras
import numpy as np
import matplotlib.pyplot as plt

(train_input, train_target), (test_input, test_target) = keras.datasets.fashion_mnist.load_data()

print(train_input.shape, train_target.shape)

fig, axs = plt.subplots(1, 10, figsize=(10, 10))
for i in range(10):
    axs[i].imshow(train_input[i], cmap='gray_r')
    axs[i].axis('off')
plt.show()

print(np.unique(train_target, return_counts=True))

"""## 인공 신경망으로 모델 만들기"""

from sklearn.model_selection import train_test_split

train_scaled = train_input / 255.0
train_scaled = train_scaled.reshape(-1, 28*28)

train_scaled, val_scaled, train_target, val_target = train_test_split(
    train_scaled, train_target, test_size=0.2, random_state=42)

dense = keras.layers.Dense(10, activation='softmax', input_shape=(784,))
model = keras.Sequential([dense])

model.compile(loss='sparse_categorical_crossentropy', metrics=['accuracy'])
model.fit(train_scaled, train_target, epochs=5)
model.evaluate(val_scaled, val_target)


"""# 심층 신경망"""
"""## 렐루 함수와 Flatten 층"""

(train_input, train_target), (test_input, test_target) = keras.datasets.fashion_mnist.load_data()
train_scaled = train_input / 255.0
train_scaled, val_scaled, train_target, val_target = train_test_split(
    train_scaled, train_target, test_size=0.2, random_state=42)

model = keras.Sequential()
model.add(keras.layers.Flatten(input_shape=(28, 28)))
model.add(keras.layers.Dense(100, activation='relu'))
model.add(keras.layers.Dense(10, activation='softmax'))

model.summary()

"""## 옵티마이저"""

model.compile(optimizer='adam', loss='sparse_categorical_crossentropy', metrics=['accuracy'])
model.fit(train_scaled, train_target, epochs=5)
model.evaluate(val_scaled, val_target)


"""# 신경망 모델 훈련"""
"""## 손실 곡선과 드롭아웃"""

def model_fn(a_layer=None):
    model = keras.Sequential()
    model.add(keras.layers.Flatten(input_shape=(28, 28)))
    model.add(keras.layers.Dense(100, activation='relu'))
    if a_layer:
        model.add(a_layer)
    model.add(keras.layers.Dense(10, activation='softmax'))
    return model

model = model_fn(keras.layers.Dropout(0.3))
model.compile(optimizer='adam', loss='sparse_categorical_crossentropy', metrics=['accuracy'])
history = model.fit(train_scaled, train_target, epochs=20, verbose=0,
                    validation_data=(val_scaled, val_target))

plt.plot(history.history['loss'])
plt.plot(history.history['val_loss'])
plt.xlabel('epoch')
plt.ylabel('loss')
plt.legend(['train', 'val'])
plt.show()

"""## 콜백 - 모델 저장과 조기 종료"""

model = model_fn(keras.layers.Dropout(0.3))
model.compile(optimizer='adam', loss='sparse_categorical_crossentropy', metrics=['accuracy'])

checkpoint_cb = keras.callbacks.ModelCheckpoint('best-model.keras', save_best_only=True)
early_stopping_cb = keras.callbacks.EarlyStopping(patience=2, restore_best_weights=True)

history = model.fit(train_scaled, train_target, epochs=20, verbose=0,
                    validation_data=(val_scaled, val_target),
                    callbacks=[checkpoint_cb, early_stopping_cb])

print(early_stopping_cb.stopped_epoch)
model.evaluate(val_scaled, val_target)
```
